  function clone(o){ return JSON.parse(JSON.stringify(o)); }

  function loadSave(){
    var out = clone(DEFAULT_SAVE);
    try{
      var raw = localStorage.getItem(SAVE_KEY);
      if(!raw) return out;
      var s = JSON.parse(raw);
      if(!s || typeof s !== "object") return out;
      if(typeof s.unlocked === "number"){
        out.unlocked = Math.min(Math.max(1, Math.floor(s.unlocked)), LEVELS.length);
      }
      if(typeof s.current === "number"){
        out.current = Math.min(Math.max(1, Math.floor(s.current)), LEVELS.length);
      }
      if(s.levels && typeof s.levels === "object"){
        for(var k in s.levels){
          if(!Object.prototype.hasOwnProperty.call(s.levels, k)) continue;
          var rec = s.levels[k];
          if(!rec || typeof rec !== "object") continue;
          out.levels[k] = {
            stars: Math.min(3, Math.max(0, Math.floor(rec.stars || 0))),
            best: Math.max(0, Math.floor(rec.best || 0))
          };
        }
      }
      if(s.settings && typeof s.settings === "object"){
        if(typeof s.settings.sound === "boolean") out.settings.sound = s.settings.sound;
        if(typeof s.settings.music === "boolean") out.settings.music = s.settings.music;
        if(typeof s.settings.vibrate === "boolean") out.settings.vibrate = s.settings.vibrate;
        if(s.settings.theme === "dark" || s.settings.theme === "light") out.settings.theme = s.settings.theme;
      }
    }catch(e){ /* corrupted save -> defaults */ }
    return out;
  }

  var Sound = {
    ctx: null,
    ensure: function(){
      if(!save.settings.sound && !save.settings.music) return null;
      if(!this.ctx){
        var AC = window.AudioContext || window.webkitAudioContext;
        if(!AC) return null;
        try{ this.ctx = new AC(); }catch(e){ return null; }
      }
      if(this.ctx.state === "suspended"){
        try{ this.ctx.resume(); }catch(e){}
      }
      return this.ctx;
    },
    blip: function(freq, dur, type, vol, delay){
      var ctx = this.ctx;
      if(!ctx) return;
      var t0 = ctx.currentTime + (delay || 0);
      var osc = ctx.createOscillator();
      var g = ctx.createGain();
      osc.type = type || "sine";
      osc.frequency.setValueAtTime(freq, t0);
      g.gain.setValueAtTime(0.0001, t0);
      g.gain.linearRampToValueAtTime(vol || 0.07, t0 + 0.012);
      g.gain.exponentialRampToValueAtTime(0.0001, t0 + dur);
      osc.connect(g);
      g.connect(ctx.destination);
      osc.start(t0);
      osc.stop(t0 + dur + 0.03);
    },
    play: function(kind){
      if(!save.settings.sound) return;
      if(!this.ensure()) return;
      switch(kind){
        case "select":
          this.blip(520, 0.07, "sine", 0.045, 0);
          break;
        case "found":
          this.blip(523.25, 0.12, "sine", 0.08, 0);
          this.blip(659.25, 0.12, "sine", 0.08, 0.09);
          this.blip(783.99, 0.20, "sine", 0.08, 0.18);
          break;
        case "wrong":
          this.blip(150, 0.18, "sawtooth", 0.05, 0);
          this.blip(110, 0.22, "sawtooth", 0.05, 0.1);
          break;
        case "win":
          this.blip(523.25, 0.16, "triangle", 0.09, 0);
          this.blip(659.25, 0.16, "triangle", 0.09, 0.14);
          this.blip(783.99, 0.16, "triangle", 0.09, 0.28);
          this.blip(1046.5, 0.34, "triangle", 0.1, 0.42);
          break;
        case "tap":
          this.blip(330, 0.05, "square", 0.03, 0);
          break;
      }
    }
  };

  function generateGrid(level){
    var size = level.size;
    var words = level.words.map(function(w){ return w.toUpperCase(); });
    words.sort(function(a,b){ return b.length - a.length; });

    for(var attempt = 0; attempt < 80; attempt++){
      var grid = [];
      for(var r = 0; r < size; r++){
        var row = [];
        for(var c = 0; c < size; c++) row.push("");
        grid.push(row);
      }
      var placements = [];
      var ok = true;

      for(var wi = 0; wi < words.length; wi++){
        var word = words[wi];
        var placed = false;

        for(var t = 0; t < 900 && !placed; t++){
          var d = DIRS[(Math.random() * 8) | 0];
          var sr = (Math.random() * size) | 0;
          var sc = (Math.random() * size) | 0;
          var er = sr + d[0] * (word.length - 1);
          var ec = sc + d[1] * (word.length - 1);
          if(er < 0 || er >= size || ec < 0 || ec >= size) continue;

          var fits = true;
          for(var i = 0; i < word.length; i++){
            var ch = grid[sr + d[0] * i][sc + d[1] * i];
            if(ch !== "" && ch !== word.charAt(i)){ fits = false; break; }
          }
          if(!fits) continue;

          for(var j = 0; j < word.length; j++){
            grid[sr + d[0] * j][sc + d[1] * j] = word.charAt(j);
          }
          placements.push({ word: word, r: sr, c: sc, dr: d[0], dc: d[1] });
          placed = true;
        }

        if(!placed){ ok = false; break; }
      }

      if(!ok) continue;

      for(var rr = 0; rr < size; rr++){
        for(var cc = 0; cc < size; cc++){
          if(grid[rr][cc] === ""){
            grid[rr][cc] = FILLER.charAt((Math.random() * 26) | 0);
          }
        }
      }
      return { grid: grid, placements: placements };
    }

    /* Extreme fallback (practically never used) */
    var fg = [];
    for(var a = 0; a < size; a++){
      var fr = [];
      for(var b = 0; b < size; b++) fr.push(FILLER.charAt((Math.random() * 26) | 0));
      fg.push(fr);
    }
    var fp = [];
    var fwords = level.words.map(function(w){ return w.toUpperCase(); });
    for(var k = 0; k < fwords.length && k < size; k++){
      var w2 = fwords[k];
      for(var m = 0; m < w2.length && m < size; m++) fg[k][m] = w2.charAt(m);
      fp.push({ word: w2, r: k, c: 0, dr: 0, dc: 1 });
    }
    return { grid: fg, placements: fp };
  }

  function computeStars(){
    var level = LEVELS[state.levelIndex];
    var target = Math.round(20 + level.words.length * 5 + level.size * level.size * 0.35);
    if(state.seconds <= target) return 3;
    if(state.seconds <= target * 1.8) return 2;
    return 1;
  }

  function levelComplete(){
    if(state.completed) return;
    state.completed = true;
    stopTimer();

    var stars = computeStars();
    state.stars = stars;

    var level = LEVELS[state.levelIndex];
    var target = Math.round(20 + level.words.length * 5 + level.size * level.size * 0.35);
    var timeBonus = Math.max(0, Math.round((target * 2 - state.seconds) * 3));
    var starBonus = stars * 100;
    state.score += timeBonus + starBonus;
    updateHUD();

    /* save progress */
    var idx = state.levelIndex + 1;
    var rec = save.levels[idx] || { stars:0, best:0 };
    if(stars > rec.stars) rec.stars = stars;
    if(state.score > rec.best) rec.best = state.score;
    save.levels[idx] = rec;

    var nextUnlock = Math.min(idx + 1, LEVELS.length);
    if(nextUnlock > save.unlocked) save.unlocked = nextUnlock;
    save.current = idx;
    persist();

    Sound.play("win");
    vibrate([60, 60, 120]);

    showLevelComplete(stars, timeBonus + starBonus);
  }

  /* =========================================================
     SELECTION LOGIC

  function lineCells(a, b){
    var dr = b.r - a.r;
    var dc = b.c - a.c;
    if(dr !== 0 && dc !== 0 && Math.abs(dr) !== Math.abs(dc)) return null;
    var stepR = dr === 0 ? 0 : (dr > 0 ? 1 : -1);
    var stepC = dc === 0 ? 0 : (dc > 0 ? 1 : -1);
    var len = Math.max(Math.abs(dr), Math.abs(dc)) + 1;
    var out = [];
    for(var i = 0; i < len; i++){
      out.push({ r: a.r + stepR * i, c: a.c + stepC * i });
    }
    return out;
  }

  function clearSelectionClasses(){
    var sel = board.querySelectorAll(".cell.sel");

  function cellFromPoint(x, y){
    var el = document.elementFromPoint(x, y);
    if(!el) return null;
    if(el.nodeType === 3) el = el.parentNode;
    if(!el || !el.closest) return null;
    var cell = el.closest(".cell");
    if(!cell || !board.contains(cell)) return null;
    return cell;
  }

  function onPointerDown(e){
    if(state.paused || state.completed) return;
    if(currentScreen !== "screen-game") return;

    var cell = cellFromPoint(e.clientX, e.clientY);
    if(!cell) return;
    if(e.preventDefault) e.preventDefault();

    activePointer = (e.pointerId === undefined) ? 1 : e.pointerId;
    clearWrong();

    var r = parseInt(cell.getAttribute("data-r"), 10);
    var c = parseInt(cell.getAttribute("data-c"), 10);

    if(state.anchor){
      if(state.anchor.r === r && state.anchor.c === c){
        state.anchor = null;
        state.selStart = null;
        state.selCells = [];
        clearSelectionClasses();
        return;
      }
      state.selStart = { r: state.anchor.r, c: state.anchor.c };
      var line = lineCells(state.selStart, { r: r, c: c });
      state.selCells = line || [state.selStart];
      paintSelection();
      return;
    }

    state.selStart = { r: r, c: c };
    state.selCells = [{ r: r, c: c }];
    state.kbSelecting = false;
    paintSelection();
    Sound.play("select");
  }

  function onPointerMove(e){
    if(activePointer === null) return;
    if(e.pointerId !== undefined && e.pointerId !== activePointer) return;
    if(!state.selStart) return;
    if(state.paused || state.completed) return;

    var cell = cellFromPoint(e.clientX, e.clientY);
    if(!cell) return;

    var r = parseInt(cell.getAttribute("data-r"), 10);
    var c = parseInt(cell.getAttribute("data-c"), 10);
    var line = lineCells(state.selStart, { r: r, c: c });
    if(line){
      state.selCells = line;
      paintSelection();
    }
  }

  function onPointerUp(e){
    if(activePointer === null) return;
    if(e.pointerId !== undefined && e.pointerId !== activePointer) return;
    activePointer = null;
    if(!state.selStart) return;
    if(state.paused || state.completed){ return; }

    if(state.selCells.length >= 2){
      var ok = checkSelection();
      if(ok){
        state.anchor = null;
        state.selStart = null;
        state.selCells = [];
      } else {
        var cells = state.selCells.slice();
        for(var i = 0; i < cells.length; i++){
          var el = state.cellEls[cells[i].r] && state.cellEls[cells[i].r][cells[i].c];
          if(el) el.classList.add("wrong");
        }
        Sound.play("wrong");
        vibrate(80);
        setTimeout(function(){
          for(var j = 0; j < cells.length; j++){
            var el2 = state.cellEls[cells[j].r] && state.cellEls[cells[j].r][cells[j].c];
            if(el2) el2.classList.remove("wrong", "sel");
          }
        }, 400);
        state.selStart = null;
        state.selCells = [];
        state.anchor = null;
      }
    } else {
      /* single letter selected -> remember as anchor for tap-tap mode */
      state.anchor = { r: state.selStart.r, c: state.selStart.c };
    }
  }

  function checkSelection(){
    var cells = state.selCells;
    if(cells.length < 2) return false;

    var str = "";
    for(var i = 0; i < cells.length; i++){
      str += state.grid[cells[i].r][cells[i].c];
    }
    var rev = str.split("").reverse().join("");

    var word = null;
    if(!state.found[str] && state.words.indexOf(str) !== -1) word = str;
    else if(!state.found[rev] && state.words.indexOf(rev) !== -1) word = rev;

    if(!word) return false;

    markWord(word);
    return true;
  }

  function markWord(word){
    var p = state.placements[word];
    if(!p) return;

    for(var i = 0; i < word.length; i++){
      var rr = p.r + p.dr * i;
      var cc = p.c + p.dc * i;
      var el = state.cellEls[rr] && state.cellEls[rr][cc];
      if(el){
        el.classList.remove("sel");
        el.classList.add("found");
      }
    }

    state.found[word] = true;

    var chip = wordListEl.querySelector('[data-word="' + word + '"]');
    if(chip) chip.classList.add("done");

    state.score += 50 + word.length * 10;
    updateHUD();
    Sound.play("found");
    vibrate(45);

    if(Object.keys(state.found).length === state.words.length){
      setTimeout(levelComplete, 500);
    }
  }

  /* =========================================================

     ========================================================= */
  function wrapTouchEvent(e){
    var t = (e.touches && e.touches[0]) || (e.changedTouches && e.changedTouches[0]);
    if(!t) return null;
    return { clientX: t.clientX, clientY: t.clientY, pointerId: 1 };
  }

  function bindPointerEvents(){
    if(window.PointerEvent){
      board.addEventListener("pointerdown", onPointerDown);
      window.addEventListener("pointermove", onPointerMove);
      window.addEventListener("pointerup", onPointerUp);
      window.addEventListener("pointercancel", onPointerUp);
    } else {
      board.addEventListener("touchstart", function(e){
        var w = wrapTouchEvent(e);
        if(w){ onPointerDown(w); }
        if(e.preventDefault) e.preventDefault();
      }, { passive:false });

      window.addEventListener("touchmove", function(e){
        var w = wrapTouchEvent(e);
        if(w) onPointerMove(w);
        if(e.preventDefault) e.preventDefault();
      }, { passive:false });

      window.addEventListener("touchend", function(e){
        var w = wrapTouchEvent(e);
        if(w) onPointerUp(w);
      });

      board.addEventListener("mousedown", onPointerDown);
      window.addEventListener("mousemove", onPointerMove);
      window.addEventListener("mouseup", onPointerUp);
    }
  }
  bindPointerEvents();

  function useHint(){
    if(state.paused || state.completed) return;
    if(state.hintsLeft <= 0){
      showToast("No hints left for this level!");
      return;
    }
    var target = null;
    for(var i = 0; i < state.words.length; i++){
      if(!state.found[state.words[i]]){ target = state.words[i]; break; }
    }
    if(!target){ showToast("All words already found!"); return; }

    var p = state.placements[target];
    if(!p) return;

    state.hintsLeft--;
    hintCountEl.textContent = state.hintsLeft;

    var el = state.cellEls[p.r] && state.cellEls[p.r][p.c];
    if(el){
      el.classList.remove("hint");
      /* force reflow so the animation restarts */
      void el.offsetWidth;
      el.classList.add("hint");
      setTimeout(function(){ el.classList.remove("hint"); }, 1700);
    }

    if(state.score >= 25) state.score -= 25; else state.score = 0;
    updateHUD();
    Sound.play("tap");
    vibrate(30);
    showToast("Hint: look for the first letter of " + target);
  }

  /* =========================================================
     SPATIAL NAVIGATION (TV / D-PAD)
     ========================================================= */
  function isVisible(el){
    if(!el) return false;
    if(el.disabled) return false;
    var r = el.getBoundingClientRect();
    if(r.width === 0 && r.height === 0) return false;
    var s = window.getComputedStyle(el);
    if(s.display === "none" || s.visibility === "hidden") return false;
    return true;
  }

  function getFocusable(){
    var all = document.querySelectorAll("button, [tabindex='0']");
    var out = [];
    for(var i = 0; i < all.length; i++){
      if(isVisible(all[i])) out.push(all[i]);
    }
    return out;
  }

  function moveFocus(dir){
    var els = getFocusable();
    if(!els.length) return false;
    var cur = document.activeElement;
    var idx = els.indexOf(cur);

    if(idx === -1){
      els[0].focus();
      return true;
    }

    var cr = cur.getBoundingClientRect();
    var cx = cr.left + cr.width / 2;
    var cy = cr.top + cr.height / 2;

    var best = null;
    var bestScore = Infinity;

    for(var i = 0; i < els.length; i++){
      var el = els[i];
      if(el === cur) continue;
      var r = el.getBoundingClientRect();
      var x = r.left + r.width / 2;
      var y = r.top + r.height / 2;
      var dx = x - cx;
      var dy = y - cy;
      var primary, secondary;

      if(dir === "ArrowUp"){
        if(dy > -6) continue;
        primary = -dy; secondary = Math.abs(dx);
      } else if(dir === "ArrowDown"){
        if(dy < 6) continue;
        primary = dy; secondary = Math.abs(dx);
      } else if(dir === "ArrowLeft"){
        if(dx > -6) continue;
        primary = -dx; secondary = Math.abs(dy);
      } else if(dir === "ArrowRight"){
        if(dx < 6) continue;
        primary = dx; secondary = Math.abs(dy);
      } else {
        continue;
      }

      var score = primary + secondary * 2.5;
      if(score < bestScore){ bestScore = score; best = el; }
    }

    if(best){
      best.focus();
      return true;
    }
    return false;
  }

  function updateCursor(){
    var all = board.querySelectorAll(".cell.cursor");
    for(var i = 0; i < all.length; i++) all[i].classList.remove("cursor");
    if(!state.cellEls.length) return;
    if(document.activeElement !== board) return;
    if(!keyboardMode) return;
    var el = state.cellEls[state.cursor.r] && state.cellEls[state.cursor.r][state.cursor.c];
    if(el) el.classList.add("cursor");
  }

  function handleGridNav(e, key){
    e.preventDefault();
    var r = state.cursor.r;
    var c = state.cursor.c;

    if(key === "ArrowUp") r = Math.max(0, r - 1);
    else if(key === "ArrowDown") r = Math.min(state.size - 1, r + 1);
    else if(key === "ArrowLeft") c = Math.max(0, c - 1);
    else if(key === "ArrowRight") c = Math.min(state.size - 1, c + 1);

    state.cursor = { r: r, c: c };

    if(state.kbSelecting && state.selStart){
      var line = lineCells(state.selStart, { r: r, c: c });
      state.selCells = line || [state.selStart];
      paintSelection();
    }
    updateCursor();
  }

  function handleGridEnter(){
    if(state.paused || state.completed) return;
    if(state.kbSelecting){
      state.kbSelecting = false;
      if(state.selCells.length >= 2){
        var ok = checkSelection();
        if(ok){
          state.selStart = null;
          state.selCells = [];
          state.anchor = null;
        } else {
          var cells = state.selCells.slice();
          for(var i = 0; i < cells.length; i++){
            var el = state.cellEls[cells[i].r] && state.cellEls[cells[i].r][cells[i].c];
            if(el) el.classList.add("wrong");
          }
          Sound.play("wrong");
          vibrate(80);
          setTimeout(function(){
            for(var j = 0; j < cells.length; j++){
              var el2 = state.cellEls[cells[j].r] && state.cellEls[cells[j].r][cells[j].c];
              if(el2) el2.classList.remove("wrong", "sel");
            }
          }, 400);
          state.selStart = null;
          state.selCells = [];
        }
      } else {
        state.selStart = null;
        state.selCells = [];
        clearSelectionClasses();
      }
    } else {
      state.kbSelecting = true;
      state.selStart = { r: state.cursor.r, c: state.cursor.c };
      state.selCells = [{ r: state.cursor.r, c: state.cursor.c }];
      state.anchor = null;
      paintSelection();
      Sound.play("select");
    }
  }

  function handleBack(){
    if(isModalOpen()){
      /* Closing via back: treat as resume if paused, otherwise just close */
      if(state.paused && currentScreen === "screen-game"){
        resumeGame();
      } else {
        closeModal();
      }
      return;
    }
    if(currentScreen === "screen-game"){
      pauseGame();
      return;
    }
    if(currentScreen !== "screen-home"){
      showScreen("screen-home");
      return;
    }
  }

  document.addEventListener("keydown", function(e){
    var key = e.key;

    /* mark keyboard mode */
    if(key !== "Unidentified"){
      if(!keyboardMode){
        keyboardMode = true;
        document.body.classList.add("kb");
      }
    }

    /* Back / Escape / Backspace */
    if(key === "Escape" || key === "Backspace" || e.keyCode === 461 || e.keyCode === 8){
      if(e.preventDefault) e.preventDefault();
      handleBack();
      return;
    }

    /* Arrow keys */
    if(key === "ArrowUp" || key === "ArrowDown" || key === "ArrowLeft" || key === "ArrowRight"){
      if(isModalOpen()) return; /* allow default button focus order inside modal */
      if(currentScreen === "screen-game" && document.activeElement === board){
        handleGridNav(e, key);
        return;
      }
      if(moveFocus(key)){
        if(e.preventDefault) e.preventDefault();
      }
      return;
    }

    /* Enter / OK */
    if(key === "Enter" || e.keyCode === 13){
      if(currentScreen === "screen-game" && document.activeElement === board){
        if(e.preventDefault) e.preventDefault();
        handleGridEnter();
      }
      return;
    }

    /* Space on board shouldn't scroll */
    if(key === " " && document.activeElement === board){
      if(e.preventDefault) e.preventDefault();
      handleGridEnter();
    }
  });

  document.addEventListener("pointerdown", function(){
    if(keyboardMode){

      var cur = board.querySelectorAll(".cell.cursor");

        document.removeEventListener("pointerdown", startOnce);
        document.removeEventListener("keydown", startOnce);
      };
      document.addEventListener("pointerdown", startOnce);
      document.addEventListener("keydown", startOnce);
    }

    /* resume audio context on first interaction */
    var unlockAudio = function(){
      Sound.ensure();
      document.removeEventListener("pointerdown", unlockAudio);
      document.removeEventListener("keydown", unlockAudio);
    };
    document.addEventListener("pointerdown", unlockAudio);
    document.addEventListener("keydown", unlockAudio);

    /* prevent double-tap zoom on buttons */
    document.addEventListener("dblclick", function(e){
      if(e.target && e.target.closest && e.target.closest("button")){
        e.preventDefault();
      }
    }, { passive:false });

    /* keep focus somewhere useful for TV on load */
    setTimeout(function(){
      var p = document.getElementById("btn-play");
      if(p) p.focus();
    }, 200);
  }

  init();

})();
